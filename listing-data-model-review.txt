// Real Estate Platform - Database Schema

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum UserRole {
  USER
  ADMIN
}

enum ListingType {
  SALE
  RENT
}

enum PropertyType {
  APARTMENT
  HOUSE
  VILLA
  LAND
  COMMERCIAL
}

enum ListingStatus {
  PENDING
  PUBLISHED
  REJECTED
  ARCHIVED
}

model User {
  sentInquiries Inquiry[] @relation("InquirySender")
  receivedInquiries Inquiry[] @relation("InquiryRecipient")
  reviews ListingReview[]
  id           String     @id @default(cuid())
  name         String?
  email        String     @unique
  phone        String?    @unique
  passwordHash String
  role         UserRole   @default(USER)

  listings     Listing[]
  favorites    Favorite[]

  createdAt    DateTime   @default(now())
  updatedAt    DateTime   @updatedAt
}

model Listing {
  inquiries Inquiry[]
  reviews ListingReview[]
  id           String        @id @default(cuid())
  title        String        @db.VarChar(120)
  description  String        @db.Text
  listingType  ListingType
  propertyType PropertyType
  status       ListingStatus @default(PENDING)

  price        Decimal       @db.Decimal(14, 2)
  currency     String        @default("RUB") @db.VarChar(3)
  area         Decimal       @db.Decimal(10, 2)
  rooms        Int?
  floor        Int?
  totalFloors  Int?

  city         String        @db.VarChar(100)
  address      String?       @db.VarChar(300)
  latitude     Float?
  longitude    Float?

  ownerId      String
  owner        User          @relation(fields: [ownerId], references: [id], onDelete: Cascade)

  images       ListingImage[]
  favorites    Favorite[]

  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt

  @@index([status, createdAt])
  @@index([city, status])
  @@index([listingType, propertyType, status])
  @@index([price])
  @@index([ownerId])
}

model ListingImage {
  id        String   @id @default(cuid())
  url       String   @db.VarChar(2048)
  sortOrder Int      @default(0)

  listingId String
  listing   Listing  @relation(fields: [listingId], references: [id], onDelete: Cascade)

  createdAt DateTime @default(now())

  @@index([listingId, sortOrder])
}

model Favorite {
  id        String   @id @default(cuid())
  userId    String
  listingId String

  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  listing   Listing  @relation(fields: [listingId], references: [id], onDelete: Cascade)

  createdAt DateTime @default(now())

  @@unique([userId, listingId])
  @@index([listingId])
}


enum ReviewDecision {
  PUBLISHED
  REJECTED
}

model ListingReview {
  id         String         @id @default(cuid())
  listingId  String
  reviewerId String?
  decision   ReviewDecision
  reason     String?        @db.VarChar(1000)
  createdAt  DateTime       @default(now())

  listing    Listing        @relation(fields: [listingId], references: [id], onDelete: Cascade)
  reviewer   User?          @relation(fields: [reviewerId], references: [id], onDelete: SetNull)

  @@index([listingId, createdAt])
  @@index([reviewerId])
}


model Inquiry {
  id          String   @id @default(cuid())
  listingId   String
  senderId    String
  recipientId String
  message     String   @db.VarChar(2000)
  isRead      Boolean  @default(false)
  createdAt   DateTime @default(now())

  listing   Listing @relation(fields: [listingId], references: [id], onDelete: Cascade)
  sender    User    @relation("InquirySender", fields: [senderId], references: [id], onDelete: Cascade)
  recipient User    @relation("InquiryRecipient", fields: [recipientId], references: [id], onDelete: Cascade)

  @@index([recipientId, createdAt])
  @@index([senderId, createdAt])

    );
  }

    return NextResponse.json(
      { error: "ابتدا وارد حساب شوید." },
      { status: 401 }
    );
  }

  const { id } = await context.params;

  if (!id || id.length > 100) {
    return NextResponse.json(
      { error: "شناسه آگهی نامعتبر است." },
      { status: 400 }
    );
  }

  const contentLength = Number(request.headers.get("content-length") || 0);

  if (contentLength > 4096) {
    return NextResponse.json(
      { error: "درخواست بیش‌ازحد بزرگ است." },
      { status: 413 }
    );
  }

  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "ساختار پیام نامعتبر است." },
      { status: 400 }
    );
  }

  if (
    !body ||
    typeof body !== "object" ||
    !("message" in body) ||
    typeof body.message !== "string"
  ) {
    return NextResponse.json(
      { error: "متن پیام الزامی است." },
      { status: 400 }
    );
  }

  const message = body.message.trim();

  if (message.length < 20 || message.length > 2000) {
    return NextResponse.json(
      { error: "متن پیام باید بین ۲۰ تا ۲۰۰۰ کاراکتر باشد." },
      { status: 400 }
    );
  }

  const listing = await prisma.listing.findFirst({
    where: { id, status: "PUBLISHED" },
    select: { id: true, ownerId: true },
  });

  if (!listing) {
    return NextResponse.json(
      { error: "آگهی در دسترس نیست." },
      { status: 404 }
    );
  }

  if (listing.ownerId === user.id) {
    return NextResponse.json(
      { error: "نمی‌توانید برای آگهی خودتان پیام بفرستید." },
      { status: 400 }
    );
  }

  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);

  const recentMessages = await prisma.inquiry.count({
    where: {
      senderId: user.id,
      createdAt: { gte: oneHourAgo },
    },
  });

  if (recentMessages >= 5) {
    return NextResponse.json(
      { error: "سقف ارسال پیام در این ساعت تکمیل شده است." },
      { status: 429 }
    );
  }

  await prisma.inquiry.create({
    data: {
      listingId: listing.id,
      senderId: user.id,
      recipientId: listing.ownerId,
      message,
    },
  });

  return NextResponse.json(
    { success: true, message: "پیام با موفقیت ارسال شد." },

  );

import { NextRequest, NextResponse } from "next/server";
import { checkRateLimit, getRequestIdentity, rateLimitResponse } from "@/src/lib/rate-limit";
import { prisma } from "@/src/lib/prisma";
import { getCurrentUser } from "@/src/lib/auth";

type Context = {
  params: Promise<{ id: string }>;
};

export async function POST(request: NextRequest, context: Context) {
  const limit = checkRateLimit("reply-inquiry", getRequestIdentity(request));
  if (!limit.allowed) {
    return rateLimitResponse(limit.retryAfterSeconds);
  }
  const origin = request.headers.get("origin");

  if (!origin || origin !== request.nextUrl.origin) {
    return NextResponse.json(
      { error: "مبدأ درخواست نامعتبر است." },
      { status: 403 }
    );
  }

  const user = await getCurrentUser();

  if (!user) {
    return NextResponse.json(
      { error: "برای پاسخ وارد حساب شوید." },
      { status: 401 }
    );
  }

  const { id } = await context.params;

  if (!id || id.length > 100) {
    return NextResponse.json(
      { error: "شناسه پیام نامعتبر است." },
      { status: 400 }
    );
  }

  const target = await prisma.inquiry.findUnique({
    where: { id },
    select: {
      id: true,
      listingId: true,
      senderId: true,
      recipientId: true,
    },
  });

  // Only the recipient of a message can reply to it.
  if (!target || target.recipientId !== user.id) {
    return NextResponse.json(
      { error: "پیام یافت نشد یا دسترسی مجاز نیست." },
      { status: 404 }
    );
  }

  if (target.senderId === user.id) {
    return NextResponse.json(
      { error: "امکان ارسال پاسخ به خودتان وجود ندارد." },
      { status: 400 }
    );
  }

  const contentType = request.headers.get("content-type") || "";

  if (!contentType.includes("application/json")) {
    return NextResponse.json(
      { error: "نوع درخواست نامعتبر است." },
      { status: 415 }
    );
  }

  const contentLength = Number(
    request.headers.get("content-length") || "0"
  );

  if (contentLength > 4096) {
    return NextResponse.json(
      { error: "درخواست بیش از حد بزرگ است." },
      { status: 413 }
    );
  }

  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "متن درخواست معتبر نیست." },
      { status: 400 }
    );
  }

  if (
    !body ||
    typeof body !== "object" ||
    !("message" in body) ||
    typeof body.message !== "string"
  ) {
    return NextResponse.json(
      { error: "متن پیام الزامی است." },
      { status: 400 }
    );
  }

  const message = body.message.trim();

  if (message.length < 2 || message.length > 2000) {
    return NextResponse.json(
      { error: "پاسخ باید بین ۲ تا ۲۰۰۰ کاراکتر باشد." },
      { status: 400 }
    );
  }

  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);

  const recentCount = await prisma.inquiry.count({
    where: {
      senderId: user.id,
      createdAt: { gte: oneHourAgo },
    },
  });

  if (recentCount >= 5) {
    return NextResponse.json(
      { error: "سقف ارسال پیام در این ساعت تکمیل شده است." },
      { status: 429 }
    );
  }

  // Preserve the original listing and reverse sender/recipient.
  const reply = await prisma.inquiry.create({
    data: {
      listingId: target.listingId,
      senderId: user.id,
      recipientId: target.senderId,
      message,
    },
    select: { id: true },
  });

  return NextResponse.json(
    { success: true, id: reply.id },
    { status: 201 }
  );
}

    );

    where: { id },
    select: {
      id: true,
      listingId: true,
      senderId: true,
      recipientId: true,
    },
  });

  if (!target || target.recipientId !== user.id) {
    return NextResponse.json(

      { status: 404 }
    );
  }

  if (target.senderId === user.id) {
    return NextResponse.json(
      { error: "امکان ارسال پاسخ به خودتان وجود ندارد." },
      { status: 400 }
    );
  }

  const contentType = request.headers.get("content-type") || "";

  if (!contentType.includes("application/json")) {
    return NextResponse.json(
      { error: "نوع درخواست نامعتبر است." },
      { status: 415 }
    );
  }

  const contentLength = Number(
    request.headers.get("content-length") || "0"
  );

  if (contentLength > 4096) {
    return NextResponse.json(
      { error: "درخواست بیش از حد بزرگ است." },
      { status: 413 }
    );
  }

  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "متن درخواست معتبر نیست." },
      { status: 400 }
    );
  }

  if (
    !body ||
    typeof body !== "object" ||
    !("message" in body) ||
    typeof body.message !== "string"
  ) {
    return NextResponse.json(
      { error: "متن پیام الزامی است." },
      { status: 400 }
    );
  }

  const message = body.message.trim();

  if (message.length < 2 || message.length > 2000) {
    return NextResponse.json(
      { error: "پاسخ باید بین ۲ تا ۲۰۰۰ کاراکتر باشد." },
      { status: 400 }
    );
  }

  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);

  const recentCount = await prisma.inquiry.count({
    where: {
      senderId: user.id,
      createdAt: { gte: oneHourAgo },
    },
  });

  if (recentCount >= 5) {
    return NextResponse.json(
      { error: "سقف ارسال پیام در این ساعت تکمیل شده است." },
      { status: 429 }
    );

  // Preserve the original listing and reverse sender/recipient.
  const reply = await prisma.inquiry.create({
    data: {
      listingId: target.listingId,

export async function PATCH(request: Request, context: Context) {
  const origin = request.headers.get("origin");
  const expectedOrigin = new URL(request.url).origin;

  if (!origin || origin !== expectedOrigin) {
    return NextResponse.json(
      { error: "مبدأ درخواست معتبر نیست." },
      { status: 403 },
    );
  }

  const user = await getCurrentUser();

  if (!user) {
    return NextResponse.json(
      { error: "ابتدا وارد حساب شوید." },
      { status: 401 },
    );
  }

  if (user.role !== "ADMIN") {
    return NextResponse.json(
      { error: "دسترسی مدیر لازم است." },
      { status: 403 },
    );
  }

  const { id } = await context.params;

  if (!id || id.length > 100) {
    return NextResponse.json(
      { error: "شناسه آگهی نامعتبر است." },
      { status: 400 },
    );
  }

  if (!(request.headers.get("content-type") ?? "")
    .toLowerCase().includes("application/json")) {
    return NextResponse.json(
      { error: "درخواست باید JSON باشد." },
      { status: 415 },
    );
  }

  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "JSON نامعتبر است." },
      { status: 400 },
    );
  }

  const parsed = decisionSchema.safeParse(body);

  if (!parsed.success) {
    return NextResponse.json(
      {
        error: "اطلاعات تصمیم مدیریت معتبر نیست.",
        details: parsed.error.flatten().fieldErrors,
      },
      { status: 400 },
    );
  }

  const { status, reason } = parsed.data;

  try {
    const result = await prisma.$transaction(async (tx) => {
      const updated = await tx.listing.updateMany({
        where: { id, status: "PENDING" },
        data: { status },
      });

      if (updated.count === 0) {
        return false;
      }

      await tx.listingReview.create({
        data: {
          listingId: id,
          reviewerId: user.id,
          decision: status,
          reason: reason || null,
        },
      });

      return true;
    });

    if (!result) {
      return NextResponse.json(
        { error: "آگهی پیدا نشد یا قبلاً بررسی شده است." },
        { status: 409 },
      );
    }

    return NextResponse.json({
      message:
        status === "PUBLISHED"
          ? "آگهی با موفقیت منتشر شد."
          : "آگهی رد شد و دلیل آن ثبت گردید.",
      id,
      status,
    });
  } catch (error) {
    console.error("Admin review failed:", error);

    return NextResponse.json(
      { error: "ثبت تصمیم مدیریت ناموفق بود." },
      { status: 500 },
    );
  }
}
