import UnreadMessagesLink from "../_components/UnreadMessagesLink";
import Link from "next/link";
import { Suspense } from "react";
import type { Prisma } from "@prisma/client";
import { prisma } from "@/src/lib/prisma";
import PropertyCard from "../_components/PropertyCard";
import "../market.css";

type RawValue = string | string[] | undefined;

type SearchParams = {
  city?: RawValue;
  deal?: RawValue;
  property?: RawValue;
  minPrice?: RawValue;
  maxPrice?: RawValue;
  minArea?: RawValue;
  maxArea?: RawValue;
  rooms?: RawValue;
  sort?: RawValue;
  page?: RawValue;
};

type SearchFilters = {
  city: string;
  deal: "SALE" | "RENT" | "";
  property: "APARTMENT" | "HOUSE" | "VILLA" | "LAND" | "COMMERCIAL" | "";
  minPrice: string;
  maxPrice: string;
  minArea: string;
  maxArea: string;
  rooms: string;
  sort: "newest" | "price_asc" | "price_desc" | "area_desc";
};

const PAGE_SIZE = 12;

const propertyTypes = [
  "APARTMENT",
  "HOUSE",
  "VILLA",
  "LAND",
  "COMMERCIAL",
] as const;

const sortOptions = [
  "newest",
  "price_asc",
  "price_desc",
  "area_desc",
] as const;

function single(value: RawValue): string {
  return typeof value === "string" ? value.trim() : "";
}

function moneyInput(value: RawValue): string {
  const raw = single(value);
  if (!/^\d{1,12}$/.test(raw)) return "";
  return String(Number(raw));
}

function areaInput(value: RawValue): string {
  const raw = single(value);
  if (!/^\d{1,8}$/.test(raw)) return "";
  return String(Number(raw));
}

function roomInput(value: RawValue): string {
  const raw = single(value);
  if (!/^\d{1,2}$/.test(raw)) return "";
  const number = Number(raw);
  return number <= 30 ? String(number) : "";
}

function parsePage(value: RawValue): number {
  const raw = single(value);

  if (!/^[1-9]\d{0,6}$/.test(raw)) {
    return 1;
  }

  return Number(raw);
}

function parseFilters(params: SearchParams): SearchFilters {
  const dealValue = single(params.deal);
  const propertyValue = single(params.property);
  const sortValue = single(params.sort);

  const deal =
    dealValue === "SALE" || dealValue === "RENT"
      ? dealValue
      : "";

  const property =
    propertyTypes.find((item) => item === propertyValue) ?? "";

  const sort =
    sortOptions.find((item) => item === sortValue) ?? "newest";

  return {
    city: single(params.city).slice(0, 100),
    deal,
    property,
    minPrice: moneyInput(params.minPrice),
    maxPrice: moneyInput(params.maxPrice),
    minArea: areaInput(params.minArea),
    maxArea: areaInput(params.maxArea),
    rooms: roomInput(params.rooms),
    sort,
  };
}

function buildWhere(filters: SearchFilters): Prisma.ListingWhereInput {
  const where: Prisma.ListingWhereInput = {
    status: "PUBLISHED",
  };

  if (filters.city) {
    where.city = {
      contains: filters.city,
      mode: "insensitive",
    };
  }

  if (filters.deal) {
    where.listingType = filters.deal;
  }

  if (filters.property) {
    where.propertyType = filters.property;
  }

  if (filters.minPrice || filters.maxPrice) {
    where.price = {
      ...(filters.minPrice ? { gte: filters.minPrice } : {}),
      ...(filters.maxPrice ? { lte: filters.maxPrice } : {}),
    };
  }

  if (filters.minArea || filters.maxArea) {
    where.area = {
      ...(filters.minArea ? { gte: filters.minArea } : {}),
      ...(filters.maxArea ? { lte: filters.maxArea } : {}),
    };
  }

  if (filters.rooms) {
    where.rooms = { gte: Number(filters.rooms) };
  }

  return where;
}

function sortBy(value: SearchFilters["sort"]): Prisma.ListingOrderByWithRelationInput[] {
  switch (value) {
    case "price_asc":
      return [{ price: "asc" }, { id: "desc" }];
    case "price_desc":
      return [{ price: "desc" }, { id: "desc" }];
    case "area_desc":
      return [{ area: "desc" }, { id: "desc" }];
    default:
      return [{ createdAt: "desc" }, { id: "desc" }];
  }
}

function pageUrl(filters: SearchFilters, page: number): string {
  const query = new URLSearchParams();

  const values: Record<string, string> = {
    city: filters.city,
    deal: filters.deal,
    property: filters.property,
    minPrice: filters.minPrice,
    maxPrice: filters.maxPrice,
    minArea: filters.minArea,
    maxArea: filters.maxArea,
    rooms: filters.rooms,
    sort: filters.sort === "newest" ? "" : filters.sort,
  };

  for (const [key, value] of Object.entries(values)) {
    if (value !== "") {
      query.set(key, value);
    }
  }

  if (page > 1) {
    query.set("page", String(page));
  }

  const encoded = query.toString();
  return encoded ? `/search?${encoded}` : "/search";
}

async function SearchResults({
  searchParams,
}: {
  searchParams: Promise<SearchParams>;
}) {
  const params = await searchParams;
  const filters = parseFilters(params);
  const where = buildWhere(filters);

  const requestedPage = parsePage(params.page);
  const total = await prisma.listing.count({ where });

  const totalPages = Math.max(1, Math.ceil(total / PAGE_SIZE));
  const currentPage = Math.min(requestedPage, totalPages);

  const listings = await prisma.listing.findMany({
    where,
    orderBy: sortBy(filters.sort),
    skip: (currentPage - 1) * PAGE_SIZE,
    take: PAGE_SIZE,
    select: {
      id: true,
      title: true,
      city: true,
      area: true,
      price: true,
      currency: true,
      rooms: true,
      propertyType: true,
      listingType: true,
      images: {
        take: 1,
        orderBy: [{ sortOrder: "asc" }, { id: "asc" }],
        select: { url: true },
      },
    },
  });

  const firstResult =
    total === 0 ? 0 : (currentPage - 1) * PAGE_SIZE + 1;

  const lastResult =
    total === 0 ? 0 : Math.min(currentPage * PAGE_SIZE, total);

  return (
    <div className="hy-advanced-layout">
      <aside className="hy-advanced-sidebar">
        <div className="hy-filter-header">
          <h2>فیلترهای جست‌وجو</h2>

            <span>شهر</span>
            <input
              name="city"
              type="search"
              defaultValue={filters.city}
              maxLength={100}
              placeholder="مثلاً مسکو"
            />
          </label>

              <option value="">خرید و اجاره</option>

            </div>
            <p className="hy-filter-hint">

function formatNumber(value: string, digits = 0) {
  const parsed = Number(value);
  return Number.isFinite(parsed)
    ? new Intl.NumberFormat("fa-IR", {
        maximumFractionDigits: digits,
      }).format(parsed)
    : value;
}

function imageIsSafe(url: string) {
  if (url.startsWith("/") && !url.startsWith("//")) {
    return true;
  }

  try {
    const parsed = new URL(url);
    return parsed.protocol === "https:" || parsed.protocol === "http:";
  } catch {
    return false;
  }
}

import { useState } from "react";

type Props = {
  title: string;

export default function PropertyGallery({ title, images }: Props) {
  const [activeIndex, setActiveIndex] = useState(0);

  if (images.length === 0) {
    return (
      <div className="hd-gallery-empty" role="img" aria-label="تصویر ملک موجود نیست">
        <span aria-hidden="true">⌂</span>
        <strong>خانه‌یار</strong>
        <p>هنوز تصویری برای این ملک ثبت نشده است.</p>
      </div>
    );
  }

  const selected = images[activeIndex] ?? images[0];

  return (
    <section className="hd-gallery" aria-label="گالری تصاویر ملک">
      <div className="hd-gallery-main">
        <Image
          key={selected}
          src={selected}
          alt={`${title} - تصویر ${activeIndex + 1}`}
          fill
          unoptimized
          priority={activeIndex === 0}
          sizes="(max-width: 900px) 100vw, 760px"
          className="hd-gallery-image"
        />

        <div className="hd-gallery-counter">
          {(activeIndex + 1).toLocaleString("fa-IR")}
          {" / "}
          {images.length.toLocaleString("fa-IR")}
        </div>

        {images.length > 1 && (
          <div className="hd-gallery-arrows">
            <button
              type="button"
              aria-label="تصویر قبلی"
              onClick={() =>
                setActiveIndex((index) =>
                  (index - 1 + images.length) % images.length
                )
              }
            >
              →
            </button>

            <button
              type="button"
              aria-label="تصویر بعدی"
              onClick={() =>
                setActiveIndex((index) => (index + 1) % images.length)
              }
            >
              ←
            </button>
          </div>
        )}
      </div>

      {images.length > 1 && (
        <div className="hd-thumbnails" aria-label="انتخاب تصویر">
          {images.map((url, index) => (
            <button
              type="button"
              key={`${url}-${index}`}
              onClick={() => setActiveIndex(index)}
              className={
                index === activeIndex
                  ? "hd-thumbnail hd-thumbnail-active"
                  : "hd-thumbnail"
              }
              aria-label={`نمایش تصویر ${index + 1}`}
              aria-pressed={index === activeIndex}
            >
              <Image
                src={url}
                alt=""
                fill

          ))}
        </div>
      )}
    </section>
  );
}

export function ShareListing() {
  const [copied, setCopied] = useState(false);

  async function share() {
    try {
      if (navigator.share) {
        await navigator.share({
          title: document.title,
          url: window.location.href,
        });
        return;
      }

      await navigator.clipboard.writeText(window.location.href);
      setCopied(true);
    } catch {
      setCopied(false);
    }
  }

  return (
    <button
      type="button"
      onClick={share}
      className="hd-share"
      aria-label="اشتراک‌گذاری لینک آگهی"
    >
      {copied ? "✓ لینک کپی شد" : "↗ اشتراک‌گذاری"}
    </button>
  );
}

"use client";

import Link from "next/link";
import { useState } from "react";

export default function InquiryForm({
  listingId,
}: {
  listingId: string;
}) {
  const [message, setMessage] = useState("");
  const [busy, setBusy] = useState(false);
  const [success, setSuccess] = useState(false);
  const [error, setError] = useState("");
  const [loginRequired, setLoginRequired] = useState(false);

  async function send(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (busy) return;

    setError("");
    setBusy(true);

    try {
      const response = await fetch(
        `/api/listings/${encodeURIComponent(listingId)}/inquiries`,
        {
          method: "POST",
          credentials: "same-origin",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ message }),
        }
      );

      if (response.status === 401) {
        setLoginRequired(true);
        return;
      }

      const data = (await response.json()) as {
        error?: string;
      };

      if (!response.ok) {
        setError(data.error || "ارسال پیام ناموفق بود.");
        return;
      }

      setSuccess(true);
      setMessage("");
    } catch {
      setError("ارتباط با سرور برقرار نشد. دوباره تلاش کنید.");
    } finally {
      setBusy(false);
    }
  }

  if (success) {
    return (
      <div className="hd-inquiry-success" role="status">
        <strong>پیامت ارسال شد.</strong>
        <p>آگهی‌دهنده می‌تواند آن را در صندوق پیام‌هایش مشاهده کند.</p>
        <Link href="/my/messages">مشاهده صندوق پیام‌ها</Link>
      </div>
    );
  }

  return (
    <form className="hd-inquiry-form" onSubmit={send}>
      <h3>پیام به آگهی‌دهنده</h3>
      <p>
        درباره شرایط بازدید یا اطلاعات بیشتر ملک سؤال بپرس.
      </p>

      <label htmlFor="inquiry-message">متن پیام</label>

      <textarea
        id="inquiry-message"
        value={message}
        onChange={(event) => setMessage(event.target.value)}
        minLength={20}
        maxLength={2000}
        required
        rows={5}
        placeholder="سلام، درباره این ملک و امکان بازدید سؤال داشتم..."
        disabled={busy || loginRequired}
      />

      <small>
        {message.length.toLocaleString("fa-IR")} / ۲۰۰۰ کاراکتر
      </small>

      {error && (
        <p className="hd-inquiry-error" role="alert">
          {error}
        </p>
      )}

      {loginRequired ? (
        <Link
          href={`/login?next=${encodeURIComponent(
            `/listings/${listingId}`
          )}`}
          className="hd-action-primary"
        >
          برای ارسال پیام وارد شوید
        </Link>
      ) : (
        <button
          type="submit"
          className="hd-action-primary"
          disabled={busy || message.trim().length < 20}
        >
          {busy ? "در حال ارسال..." : "ارسال پیام"}
        </button>
      )}
    </form>
  );
}
