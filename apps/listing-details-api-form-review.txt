import { NextResponse } from "next/server";
import { checkRateLimit, getRequestIdentity, rateLimitResponse } from "../../../src/lib/rate-limit";
import { prisma } from "../../../src/lib/prisma";
import { getCurrentUser } from "../../../src/lib/auth";
import { listingSchema } from "@real-estate/shared";

export async function POST(request: Request) {
  const limit = await checkRateLimit("create-listing", getRequestIdentity(request));
  if (!limit.allowed) {
    return rateLimitResponse(limit.retryAfterSeconds);
  }
  try {
    // Prevent cross-origin browser submissions.
    const origin = request.headers.get("origin");
    const requestOrigin = new URL(request.url).origin;

    if (!origin || origin !== requestOrigin) {
      return NextResponse.json(
        { error: "مبدأ درخواست معتبر نیست." },
        { status: 403 },
      );
    }

    // Ownership must come from the verified session.
    const user = await getCurrentUser();

    if (!user) {
      return NextResponse.json(
        { error: "برای ثبت آگهی ابتدا وارد حساب شوید." },
        { status: 401 },
      );
    }

    const contentType = request.headers.get("content-type") ?? "";

    if (!contentType.toLowerCase().startsWith("multipart/form-data")) {
      return NextResponse.json(
        { error: "فرمت درخواست باید FormData باشد." },
        { status: 415 },
      );
    }

    let formData: FormData;

    try {
      formData = await request.formData();
    } catch {
      return NextResponse.json(
        { error: "اطلاعات ارسال‌شده قابل خواندن نیست." },
        { status: 400 },
      );
    }

    const rawInput = {
      title: String(formData.get("title") ?? ""),
      description: String(formData.get("description") ?? ""),
      listingType: String(formData.get("listingType") ?? ""),
      propertyType: String(formData.get("propertyType") ?? ""),
      price: Number(formData.get("price")),
      currency: String(formData.get("currency") ?? "RUB"),
      area: Number(formData.get("area")),
      rooms:
        formData.get("rooms") === null || formData.get("rooms") === ""
          ? null
          : Number(formData.get("rooms")),
      city: String(formData.get("city") ?? ""),
      address: String(formData.get("address") ?? "").trim() || null,
      latitude: null,
      longitude: null,
      imageUrls: [],
    };

    const validation = listingSchema.safeParse(rawInput);

    if (!validation.success) {
      return NextResponse.json(
        {
          error: "اطلاعات فرم معتبر نیست.",
          details: validation.error.flatten(),
        },
        { status: 400 },
      );
    }

    const floorValue = formData.get("floor");
    const totalFloorsValue = formData.get("totalFloors");

    const floor =
      floorValue === null || floorValue === ""
        ? null
        : Number(floorValue);

    const totalFloors =
      totalFloorsValue === null || totalFloorsValue === ""
        ? null
        : Number(totalFloorsValue);

    if (
      (floor !== null && (!Number.isInteger(floor) || floor < 0)) ||
      (totalFloors !== null &&
        (!Number.isInteger(totalFloors) || totalFloors < 1)) ||
      (floor !== null &&
        totalFloors !== null &&
        floor > totalFloors)
    ) {
      return NextResponse.json(
        { error: "اطلاعات طبقه معتبر نیست." },
        { status: 400 },
      );
    }

    const listing = await prisma.listing.create({
      data: {
        title: validation.data.title,
        description: validation.data.description,
        listingType: validation.data.listingType,
        propertyType: validation.data.propertyType,
        price: validation.data.price,
        currency: validation.data.currency,
        area: validation.data.area,
        rooms: validation.data.rooms,
        city: validation.data.city,
        address: validation.data.address,
        latitude: validation.data.latitude,
        longitude: validation.data.longitude,
        floor,
        totalFloors,
        ownerId: user.id,
      },
      select: {
        id: true,
        title: true,
        status: true,
        ownerId: true,
        createdAt: true,
      },
    });

    return NextResponse.json(
      {
        message: "آگهی ثبت شد و منتظر تأیید مدیر است.",
        listing,
      },
      { status: 201 },
    );
  } catch (error) {
    console.error("Create listing failed:", error);

    return NextResponse.json(
      { error: "خطای داخلی هنگام ثبت آگهی." },
      { status: 500 },
    );
  }
}

    );

    return NextResponse.json(
      { error: "اطلاعات درخواست معتبر نیست." },
      { status: 400 },
    );
  }

  const input = body as Record<string, unknown>;
  const allowedKeys = new Set([
    "title", "description", "listingType", "propertyType",
    "price", "currency", "area", "rooms", "city", "address",
    "latitude", "longitude", "floor", "totalFloors",
  ]);

  if (Object.keys(input).some((key) => !allowedKeys.has(key))) {
    return NextResponse.json(
      { error: "فیلد غیرمجاز در درخواست وجود دارد." },
      { status: 400 },
    );
  }

  const validation = listingSchema.safeParse({
    ...input,
    imageUrls: [],
  });

  if (!validation.success) {
    return NextResponse.json(
      {
        error: "اطلاعات فرم معتبر نیست.",
        details: validation.error.flatten(),
      },
      { status: 400 },
    );
  }

  const { floor, totalFloors } = input;

  const validFloor =
    floor === null ||
    (typeof floor === "number" && Number.isInteger(floor) && floor >= 0);

    totalFloors === null ||
    (typeof totalFloors === "number" &&
      Number.isInteger(totalFloors) && totalFloors >= 1);

  if (
    !validFloor ||
    !validTotalFloors ||
    floor === undefined ||
    totalFloors === undefined ||
    (typeof floor === "number" &&
      typeof totalFloors === "number" &&
      floor > totalFloors)
  ) {
    return NextResponse.json(
      { error: "اطلاعات طبقه معتبر نیست." },
      { status: 400 },
    );
  }

  const data = validation.data;

    where: { id, ownerId: user.id },

      rooms: data.rooms,
      city: data.city,
      address: data.address,
      latitude: data.latitude,
      longitude: data.longitude,

import { NextRequest, NextResponse } from "next/server";
import { checkRateLimit, getRequestIdentity, rateLimitResponse } from "@/src/lib/rate-limit";
import { prisma } from "@/src/lib/prisma";
import { getCurrentUser } from "@/src/lib/auth";

type Context = {
  params: Promise<{ id: string }>;
};

export async function POST(request: NextRequest, context: Context) {
  const limit = await checkRateLimit(
    "create-inquiry",
    getRequestIdentity(request),
  );

  if (!limit.allowed) {
    return rateLimitResponse(limit.retryAfterSeconds);
  }
  const origin = request.headers.get("origin");

  if (origin !== request.nextUrl.origin) {
    return NextResponse.json(
      { error: "مبدأ درخواست معتبر نیست." },
      { status: 403 }
    );
  }

  const user = await getCurrentUser();

  if (!user) {
    return NextResponse.json(
      { error: "ابتدا وارد حساب شوید." },
      { status: 401 }
    );
  }

  const { id } = await context.params;

  if (!id || id.length > 100) {
    return NextResponse.json(
      { error: "شناسه آگهی نامعتبر است." },
      { status: 400 }
    );
  }

  const contentLength = Number(request.headers.get("content-length") || 0);

  if (contentLength > 4096) {
    return NextResponse.json(
      { error: "درخواست بیش‌ازحد بزرگ است." },
      { status: 413 }
    );
  }

  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "ساختار پیام نامعتبر است." },
      { status: 400 }
    );
  }

  if (
    !body ||
    typeof body !== "object" ||
    !("message" in body) ||
    typeof body.message !== "string"
  ) {
    return NextResponse.json(
      { error: "متن پیام الزامی است." },
      { status: 400 }
    );
  }

  const message = body.message.trim();

  if (message.length < 20 || message.length > 2000) {
    return NextResponse.json(
      { error: "متن پیام باید بین ۲۰ تا ۲۰۰۰ کاراکتر باشد." },
      { status: 400 }
    );
  }

  const listing = await prisma.listing.findFirst({
    where: { id, status: "PUBLISHED" },
    select: { id: true, ownerId: true },
  });

  if (!listing) {
    return NextResponse.json(
      { error: "آگهی در دسترس نیست." },
      { status: 404 }
    );
  }

  if (listing.ownerId === user.id) {
    return NextResponse.json(
      { error: "نمی‌توانید برای آگهی خودتان پیام بفرستید." },
      { status: 400 }
    );
  }

  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);

  const recentMessages = await prisma.inquiry.count({
    where: {
      senderId: user.id,
      createdAt: { gte: oneHourAgo },
    },
  });

  if (recentMessages >= 5) {
    return NextResponse.json(
      { error: "سقف ارسال پیام در این ساعت تکمیل شده است." },
      { status: 429 }
    );
  }

  await prisma.inquiry.create({
    data: {
      listingId: listing.id,
      senderId: user.id,
      recipientId: listing.ownerId,
      message,
    },
  });

  return NextResponse.json(
    { success: true, message: "پیام با موفقیت ارسال شد." },
    { status: 201 }
  );
}

  listingId,

  const [loginRequired, setLoginRequired] = useState(false);

  async function send(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (busy) return;

    setError("");
    setBusy(true);

    try {
      const response = await fetch(
        `/api/listings/${encodeURIComponent(listingId)}/inquiries`,
        {
          method: "POST",
          credentials: "same-origin",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ message }),
        }
      );

      if (response.status === 401) {
        setLoginRequired(true);
        return;
      }

      const data = (await response.json()) as {
        error?: string;
      };

      if (!response.ok) {
        setError(data.error || "ارسال پیام ناموفق بود.");
        return;
      }

      setSuccess(true);
      setMessage("");
    } catch {
      setError("ارتباط با سرور برقرار نشد. دوباره تلاش کنید.");
    } finally {
      setBusy(false);
    }
  }

  if (success) {
    return (
      <div className="hd-inquiry-success" role="status">
        <strong>پیامت ارسال شد.</strong>
        <p>آگهی‌دهنده می‌تواند آن را در صندوق پیام‌هایش مشاهده کند.</p>

    );

    <form className="hd-inquiry-form" onSubmit={send}>
      <h3>پیام به آگهی‌دهنده</h3>

type Props = { title: string; images: string[] };

export default function PropertyGallery({ title, images }: Props) {
  const [activeIndex, setActiveIndex] = useState(0);
  const [lightboxOpen, setLightboxOpen] = useState(false);

  const closeLightbox = useCallback(() => setLightboxOpen(false), []);
  const previous = useCallback(() => setActiveIndex((index) => (index - 1 + images.length) % images.length), [images.length]);
  const next = useCallback(() => setActiveIndex((index) => (index + 1) % images.length), [images.length]);

  useEffect(() => {
    if (!lightboxOpen) return;
    const onKeyDown = (event: KeyboardEvent) => {
      if (event.key === "Escape") closeLightbox();
      if (event.key === "ArrowLeft") next();
      if (event.key === "ArrowRight") previous();
    };
    window.addEventListener("keydown", onKeyDown);
    document.body.style.overflow = "hidden";
    return () => {
      window.removeEventListener("keydown", onKeyDown);
      document.body.style.overflow = "";
    };
  }, [lightboxOpen, closeLightbox, next, previous]);

  if (images.length === 0) {
    return <div className="hd-gallery-empty" role="img" aria-label="تصویر ملک موجود نیست"><span aria-hidden="true">⌂</span><strong>خانه‌یار</strong><p>هنوز تصویری برای این ملک ثبت نشده است.</p></div>;
  }

  const selected = images[activeIndex] ?? images[0];

  return <>
    <section className="hd-gallery" aria-label="گالری تصاویر ملک">
      <div className="hd-gallery-main">
        <button type="button" className="hd-gallery-open" onClick={() => setLightboxOpen(true)} aria-label="نمایش تصویر در اندازه بزرگ">

        {images.length > 1 && <div className="hd-gallery-arrows">
          <button type="button" aria-label="تصویر قبلی" onClick={previous}>→</button>

        </div>}
      </div>
      {images.length > 1 && <div className="hd-thumbnails" aria-label="انتخاب تصویر">
        {images.map((url, index) => <button type="button" key={`${url}-${index}`} onClick={() => setActiveIndex(index)} className={index === activeIndex ? "hd-thumbnail hd-thumbnail-active" : "hd-thumbnail"} aria-label={`نمایش تصویر ${index + 1}`} aria-pressed={index === activeIndex}>

    </section>

export function ShareListing() {
  const [copied, setCopied] = useState(false);
  async function share() {
    try {
      if (navigator.share) {
        await navigator.share({ title: document.title, url: window.location.href });
        return;
      }
      await navigator.clipboard.writeText(window.location.href);
      setCopied(true);
      window.setTimeout(() => setCopied(false), 2200);
    } catch { setCopied(false); }
  }
  return <button type="button" onClick={share} className="hd-share" aria-label="اشتراک‌گذاری لینک آگهی">{copied ? "✓ لینک کپی شد" : "↗ اشتراک‌گذاری"}</button>;
